![Screenshot 2026-03-08 at 7.09.06 pm](assets/Screenshot%202026-03-08%20at%207.09.06%E2%80%AFpm.png)

```js
/**
 * @param {number[]} nums
 * @return {number}
 */
var singleNumber = function (nums) {
  let ans = 0
  for (const x of nums) {
    ans = ans ^ x
  }
  return ans
}
```
